cmake_minimum_required(VERSION 3.28)
project(workshop_t02 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(WORKSHOP_VENDOR "" CACHE PATH "Read-only verified vendor source root")
set(SANITIZER "" CACHE STRING "address,undefined or thread")
if(NOT EXISTS "${WORKSHOP_VENDOR}/drogon/CMakeLists.txt" OR NOT EXISTS "${WORKSHOP_VENDOR}/sqlite/sqlite3.c" OR NOT EXISTS "${WORKSHOP_VENDOR}/cmark/CMakeLists.txt")
  message(FATAL_ERROR "Missing pinned source closure; no downloads allowed")
endif()
if(SANITIZER)
  add_compile_options(-fsanitize=${SANITIZER} -fno-omit-frame-pointer -g)
  add_link_options(-fsanitize=${SANITIZER})
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(BUILD_CTL OFF CACHE BOOL "" FORCE)
set(BUILD_ORM OFF CACHE BOOL "" FORCE)
set(BUILD_BROTLI OFF CACHE BOOL "" FORCE)
set(BUILD_YAML_CONFIG OFF CACHE BOOL "" FORCE)
set(BUILD_C-ARES OFF CACHE BOOL "" FORCE)
set(TRANTOR_USE_TLS openssl CACHE STRING "" FORCE)
set(USE_SUBMODULE ON CACHE BOOL "" FORCE)
set(CMARK_TESTS OFF CACHE BOOL "" FORCE)
add_subdirectory("${WORKSHOP_VENDOR}/drogon" vendor/drogon EXCLUDE_FROM_ALL SYSTEM)
add_subdirectory("${WORKSHOP_VENDOR}/cmark" vendor/cmark EXCLUDE_FROM_ALL SYSTEM)
find_package(Threads REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(JSONCPP REQUIRED IMPORTED_TARGET jsoncpp)
add_library(workshop_sqlite STATIC "${WORKSHOP_VENDOR}/sqlite/sqlite3.c")
target_include_directories(workshop_sqlite SYSTEM PUBLIC "${WORKSHOP_VENDOR}/sqlite")
target_compile_definitions(workshop_sqlite PRIVATE SQLITE_THREADSAFE=1 SQLITE_DQS=0 SQLITE_DEFAULT_FOREIGN_KEYS=1)
target_link_libraries(workshop_sqlite PUBLIC Threads::Threads ${CMAKE_DL_LIBS})
foreach(lib workshop_core workshop_test_core)
  add_library(${lib} src/model.cpp src/store.cpp src/executor.cpp src/updates.cpp src/telemetry.cpp)
  target_include_directories(${lib} PUBLIC include)
  target_include_directories(${lib} SYSTEM PUBLIC "${WORKSHOP_VENDOR}/cmark/src" "${CMAKE_CURRENT_BINARY_DIR}/vendor/cmark/src")
  target_link_libraries(${lib} PUBLIC workshop_sqlite cmark PkgConfig::JSONCPP Threads::Threads)
  target_compile_options(${lib} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wsign-conversion -Werror)
endforeach()
target_compile_definitions(workshop_test_core PUBLIC WORKSHOP_TESTING)
add_executable(workshop_backend src/main.cpp src/update_transport.cpp)
target_link_libraries(workshop_backend PRIVATE workshop_core drogon)
add_executable(workshop_tests tests/tests.cpp)
target_link_libraries(workshop_tests PRIVATE workshop_test_core)
target_compile_definitions(workshop_tests PRIVATE WORKSHOP_SCHEMA="${CMAKE_CURRENT_SOURCE_DIR}/schema/001.sql")
foreach(target workshop_backend workshop_tests)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wsign-conversion -Werror)
endforeach()
enable_testing()
include(tests/cases.cmake)

if(SANITIZER STREQUAL "thread")
  add_executable(workshop_tsan_control ../tests/t02/sqlite-diagnostic.cpp)
  target_link_libraries(workshop_tsan_control PRIVATE workshop_sqlite)
  target_compile_options(workshop_tsan_control PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()

add_executable(workshop_updates_tests tests/updates.cpp)
target_link_libraries(workshop_updates_tests PRIVATE workshop_test_core)
target_compile_definitions(workshop_updates_tests PRIVATE WORKSHOP_SCHEMA="${CMAKE_CURRENT_SOURCE_DIR}/schema/001.sql")
target_compile_options(workshop_updates_tests PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wsign-conversion -Werror)

add_executable(workshop_telemetry_tests tests/telemetry.cpp)
target_link_libraries(workshop_telemetry_tests PRIVATE workshop_test_core)
target_compile_definitions(workshop_telemetry_tests PRIVATE WORKSHOP_SCHEMA="${CMAKE_CURRENT_SOURCE_DIR}/schema/001.sql")
target_compile_options(workshop_telemetry_tests PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wsign-conversion -Werror)
