cmake_minimum_required(VERSION 3.28)
project(kidea_r05_sample LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(SANITIZER "" CACHE STRING "address,undefined or thread")
if(SANITIZER)
  add_compile_options(-fsanitize=${SANITIZER} -fno-omit-frame-pointer -g)
  add_link_options(-fsanitize=${SANITIZER})
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(BUILD_CTL OFF CACHE BOOL "" FORCE)
set(BUILD_ORM OFF CACHE BOOL "" FORCE)
set(BUILD_BROTLI OFF CACHE BOOL "" FORCE)
set(BUILD_YAML_CONFIG OFF CACHE BOOL "" FORCE)
set(BUILD_C-ARES OFF CACHE BOOL "" FORCE)
set(TRANTOR_USE_TLS openssl CACHE STRING "" FORCE)
set(USE_SUBMODULE ON CACHE BOOL "" FORCE)
add_subdirectory(vendor/drogon EXCLUDE_FROM_ALL SYSTEM)
set(CMARK_TESTS OFF CACHE BOOL "" FORCE)
add_subdirectory(vendor/cmark EXCLUDE_FROM_ALL SYSTEM)
find_package(Threads REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(JSONCPP REQUIRED IMPORTED_TARGET jsoncpp)
add_library(sqlite_amalgamation STATIC vendor/sqlite/sqlite3.c)
target_include_directories(sqlite_amalgamation PUBLIC vendor/sqlite)
target_compile_definitions(sqlite_amalgamation PRIVATE SQLITE_THREADSAFE=1 SQLITE_DQS=0 SQLITE_DEFAULT_FOREIGN_KEYS=1)
target_link_libraries(sqlite_amalgamation PUBLIC Threads::Threads ${CMAKE_DL_LIBS})
add_library(domain cpp/domain.cpp cpp/input.cpp)
target_include_directories(domain PUBLIC cpp vendor/cmark/src ${CMAKE_CURRENT_BINARY_DIR}/vendor/cmark/src)
target_link_libraries(domain PUBLIC sqlite_amalgamation cmark PkgConfig::JSONCPP)
target_compile_options(domain PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wsign-conversion -Werror)
add_executable(domain_tests cpp/tests.cpp)
target_compile_options(domain_tests PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wsign-conversion -Werror)
target_link_libraries(domain_tests PRIVATE domain Threads::Threads)
add_executable(backend cpp/server.cpp)
target_compile_options(backend PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wsign-conversion -Werror)
target_link_libraries(backend PRIVATE domain drogon)
enable_testing()
foreach(case grammar unicode atomicity rollback sql concurrency crash lifetime bounded_queue)
 add_test(NAME ${case} COMMAND domain_tests ${case})
 set_tests_properties(${case} PROPERTIES TIMEOUT 20)
endforeach()
